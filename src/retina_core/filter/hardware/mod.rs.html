25// priority levels for ingress rules
26const HIGH_PRIORITY: u32 = 0;
27const LOW_PRIORITY: u32 = 3;
28
29#[derive(Debug)]
30pub(crate) struct HardwareFilter<'a> {
31    patterns: Vec<LayeredPattern>,
32    port: &'a Port,
33}
34
35impl<'a> HardwareFilter<'a> {
36    /// Creates a new HardwareFilter for port given a filter.
37    /// Prunes all predicates not supported by the device.
38    pub(crate) fn new(filter: &Filter, port: &'a Port) -> Self {
39        let hw_patterns = filter
40            .get_patterns_flat()
41            .iter()
42            .map(|p| p.retain_hardware_predicates(port))
43            .collect::<Vec<_>>();
44
45        // Prune some hidden (redundant) patterns.
46        // Only removes those with same prefix
47        let mut hw_ptree = PTree::new(&hw_patterns);
48        hw_ptree.prune_branches();
49        let mut hw_patterns = hw_ptree.to_flat_patterns();
50
51        let mut layered = vec![];
52        for pattern in hw_patterns.iter_mut() {
53            // Only retaining hardware filterable predicates may make
54            // a pattern no-longer fully qualified. Therefore, we must
55            // broaden pattern until hardware filterable again
56            while !pattern.is_fully_qualified() {
57                pattern.predicates.pop();
58            }
59            // converts to LayeredPattern
60            layered.extend(pattern.to_fully_qualified().expect("fully qualified"));
61        }
62
63        // Remove identical patterns
64        layered.sort();
65        layered.dedup();
66
67        HardwareFilter {
68            patterns: layered,
69            port,
70        }
71    }
72
73    /// Installs the hardware filter to the port.
74    pub(crate) fn install(&self) -> Result<()> {
75        debug!("{}", self);
76        if self.patterns.iter().all(|p| p.is_empty()) {
77            info!("Empty filter, skipping.");
78            return Ok(());
79        }
80
81        info!("Applying hardware filter rules on Port {}...", self.port.id);
82        for pattern in self.patterns.iter() {
83            install_pattern(pattern, self.port, 0, HIGH_PRIORITY)?;
84        }
85        // Non-matching traffic will be dropped by default on table 1
86        // Redirect is faster than using a default DROP rule
87        add_redirect(self.port, 0, 1, LOW_PRIORITY)?;
88        // drop_eth_traffic(self.port, 0, LOW_PRIORITY)?;
89
90        Ok(())
91    }
92}

122pub(crate) fn device_supported(pred: &Predicate, port: &Port) -> bool {
123    // Device supported protocols
124    let hw_filterable_protos = hashset! {
125        protocol!("ipv4"),
126        protocol!("ipv6"),
127        protocol!("tcp"),
128        protocol!("udp"),
129    };
130    let proto_supported = hw_filterable_protos.contains(pred.get_protocol());
131    if !proto_supported {
132        info!("Hardware filter does not support protocol for: [{}]", pred);
133        return false;
134    }
135
136    // Only allow equality predicates
137    // MLX5 only supports equality or masked IP address
138    let op_supported = match pred {
139        Predicate::Unary { .. } => true,
140        Predicate::Binary {
141            protocol,
142            field: _,
143            op,
144            value: _,
145        } => {
146            matches!(op, BinOp::Eq)
147                || protocol == &protocol!("ipv4") && matches!(op, BinOp::In)
148                || protocol == &protocol!("ipv6") && matches!(op, BinOp::In)
149        }
150    };
151    if !op_supported {
152        info!(
153            "Hardware filter does not support binary comparison operator for: [{}]",
154            pred
155        );
156        return false;
157    }
158
159    // The only way to truly tell if predicate on a field is supported is to
160    // fully-qualify it and test if all fully-qualified patterns validate successfully.
161    // This still does not guarantee the flow rule will create successfully.
162    // For example, a collision detected or device resource limitations.
163    // Hardware Rules are created on table 0 (group 0) with high priority
164    // matching.
165    let pred_supported = predicate_supported(pred, port, 0, HIGH_PRIORITY);
166    if !pred_supported {
167        info!("Hardware filter does not support predicate: [{}]", pred);
168        return false;
169    }
170    true
171}

173fn predicate_supported(predicate: &Predicate, port: &Port, group: u32, priority: u32) -> bool {
174    let pattern = FlatPattern {
175        predicates: vec![predicate.to_owned()],
176    };
177    let fq_patterns = pattern.to_fully_qualified().expect("fully qualified");
178    fq_patterns
179        .iter()
180        .all(|p| pattern_supported(p, port, group, priority))
181}

183fn pattern_supported(lpattern: &LayeredPattern, port: &Port, group: u32, priority: u32) -> bool {
184    let attr = FlowAttribute::new(group, priority);
185    let pattern = FlowPattern::from_layered_pattern(lpattern);
186    // debug!("lpattern: {}", lpattern);
187    match pattern {
188        Ok(mut pattern) => {
189            let mut action = FlowAction::new(port.id);
190
191            action.append_rss();
192            action.finish();
193
194            validate_rule(port, attr, &mut pattern, &mut action)
195        }
196        Err(error) => {
197            warn!("{}", error);
198            false
199        }
200    }
201}

203fn validate_rule(
204    port: &Port,
205    attr: FlowAttribute,
206    pattern: &mut FlowPattern,
207    action: &mut FlowAction,
208) -> bool {
209    let mut pattern_rules: PatternRules = vec![];
210    flow_item::append_eth(&mut pattern_rules);
211    for item in pattern.items.iter() {
212        let mut p_item: dpdk::rte_flow_item = unsafe { mem::zeroed() };
213        p_item.type_ = item.item_type();
214        p_item.spec = item.spec();
215        p_item.mask = item.mask();
216        pattern_rules.push(p_item);
217    }
218    flow_item::append_end(&mut pattern_rules);
219
220    // Need to update flow_action_rss here
221    // reta_raw needs to stay in scope until after rte_flow_validate() succeeds
222    let reta_raw = port.reta.iter().map(|q| q.raw()).collect::<Vec<_>>();
223    for a in action.rules.iter_mut() {
224        if let dpdk::rte_flow_action_type_RTE_FLOW_ACTION_TYPE_RSS = a.type_ {
225            action.rss[0].queue_num = port.queue_map.len() as u32;
226            action.rss[0].queue = reta_raw.as_ptr();
227            a.conf = &action.rss[0] as *const _ as *const c_void;
228        }
229    }
230
231    let mut error: dpdk::rte_flow_error = unsafe { mem::zeroed() };
232    unsafe {
233        let ret = dpdk::rte_flow_validate(
234            port.id.raw(),
235            attr.raw() as *const _,
236            pattern_rules.as_ptr(),
237            action.rules.as_ptr(),
238            &mut error as *mut _,
239        );
240        ret == 0
241    }
242}

244fn install_pattern(
245    lpattern: &LayeredPattern,
246    port: &Port,
247    group: u32,
248    priority: u32,
249) -> Result<()> {
250    let attr = FlowAttribute::new(group, priority);
251    if let Ok(mut pattern) = FlowPattern::from_layered_pattern(lpattern) {
252        let mut action = FlowAction::new(port.id);
253        // action.append_mark(tag as u32);
254
255        action.append_rss();
256        action.finish();
257
258        create_rule(lpattern, port, attr, &mut pattern, &mut action)
259    } else {
260        bail!(HardwareFilterError::InvalidRule(lpattern.to_owned()));
261    }
262}

264fn create_rule(
265    lpattern: &LayeredPattern,
266    port: &Port,
267    attr: FlowAttribute,
268    pattern: &mut FlowPattern,
269    action: &mut FlowAction,
270) -> Result<()> {
271    let mut pattern_rules: PatternRules = vec![];
272    flow_item::append_eth(&mut pattern_rules);
273    for item in pattern.items.iter() {
274        let mut p_item: dpdk::rte_flow_item = unsafe { mem::zeroed() };
275        p_item.type_ = item.item_type();
276        p_item.spec = item.spec();
277        p_item.mask = item.mask();
278        pattern_rules.push(p_item);
279    }
280    flow_item::append_end(&mut pattern_rules);
281
282    // Need to update flow_action_rss here
283    // reta_raw needs to stay in scope until after rte_flow_create() succeeds
284    let reta_raw = port.reta.iter().map(|q| q.raw()).collect::<Vec<_>>();
285    for a in action.rules.iter_mut() {
286        if let dpdk::rte_flow_action_type_RTE_FLOW_ACTION_TYPE_RSS = a.type_ {
287            action.rss[0].queue_num = port.queue_map.len() as u32;
288            action.rss[0].queue = reta_raw.as_ptr();
289            a.conf = &action.rss[0] as *const _ as *const c_void;
290        }
291    }
292
293    let mut error: dpdk::rte_flow_error = unsafe { mem::zeroed() };
294    unsafe {
295        let ret = dpdk::rte_flow_validate(
296            port.id.raw(),
297            attr.raw() as *const _,
298            pattern_rules.as_ptr(),
299            action.rules.as_ptr(),
300            &mut error as *mut _,
301        );
302        if ret != 0 {
303            let msg: &CStr = CStr::from_ptr(error.message);
304            bail!(HardwareFilterError::Validation {
305                lpattern: lpattern.to_owned(),
306                reason: msg.to_str().unwrap().to_string()
307            });
308        } else {
309            let ret = dpdk::rte_flow_create(
310                port.id.raw(),
311                attr.raw() as *const _,
312                pattern_rules.as_ptr(),
313                action.rules.as_ptr(),
314                &mut error as *mut _,
315            );
316            if ret.is_null() {
317                let msg: &CStr = CStr::from_ptr(error.message);
318                bail!(HardwareFilterError::Creation {
319                    lpattern: lpattern.to_owned(),
320                    reason: msg.to_str().unwrap().to_string()
321                });
322            } else {
323                info!("Hardware flow rule created: {}", lpattern);
324                Ok(())
325            }
326        }
327    }
328}

330fn add_redirect(port: &Port, from_group: u32, to_group: u32, priority: u32) -> Result<()> {
331    let attr = FlowAttribute::new(from_group, priority);
332
333    // Pattern matches all Ethernet traffic
334    let mut pattern_rules: PatternRules = vec![];
335    flow_item::append_eth(&mut pattern_rules);
336    flow_item::append_end(&mut pattern_rules);
337
338    // Set action to redirect
339    let mut action = FlowAction::new(port.id);
340    action.append_jump(to_group);
341    action.finish();
342    for a in action.rules.iter_mut() {
343        if let dpdk::rte_flow_action_type_RTE_FLOW_ACTION_TYPE_JUMP = a.type_ {
344            a.conf = &action.jump[0] as *const _ as *const c_void;
345        }
346    }
347
348    info!(
349        "Setting port {} to redirect from group {} to {}...",
350        port.id, from_group, to_group
351    );
352
353    let mut error: dpdk::rte_flow_error = unsafe { mem::zeroed() };
354    unsafe {
355        let ret = dpdk::rte_flow_validate(
356            port.id.raw(),
357            attr.raw() as *const _,
358            pattern_rules.as_ptr(),
359            action.rules.as_ptr(),
360            &mut error as *mut _,
361        );
362        if ret != 0 {
363            let msg: &CStr = CStr::from_ptr(error.message);
364            error!("Redirect rule failed validation: {}", msg.to_str().unwrap());
365            bail!(HardwareFilterError::Validation {
366                lpattern: LayeredPattern::new(),
367                reason: msg.to_str().unwrap().to_string()
368            });
369        } else {
370            let ret = dpdk::rte_flow_create(
371                port.id.raw(),
372                attr.raw() as *const _,
373                pattern_rules.as_ptr(),
374                action.rules.as_ptr(),
375                &mut error as *mut _,
376            );
377            if ret.is_null() {
378                error!("Redirect rule failed creation.");
379                let msg: &CStr = CStr::from_ptr(error.message);
380                bail!(HardwareFilterError::Creation {
381                    lpattern: LayeredPattern::new(),
382                    reason: msg.to_str().unwrap().to_string()
383                });
384            } else {
385                info!("Created hardware flow rule for redirect.");
386            }
387        }
388    }
389    Ok(())
390}

392#[allow(dead_code)]
393fn drop_eth_traffic(port: &Port, group: u32, priority: u32) -> Result<()> {
394    let attr = FlowAttribute::new(group, priority);
395
396    // Pattern matches all Ethernet traffic
397    let mut pattern_rules: PatternRules = vec![];
398    flow_item::append_eth(&mut pattern_rules);
399    flow_item::append_end(&mut pattern_rules);
400
401    // Set action to DROP
402    let mut action = FlowAction::new(port.id);
403    action.append_drop();
404    action.finish();
405
406    info!(
407        "Setting port {} to drop all ethernet traffic by default...",
408        port.id
409    );
410
411    let mut error: dpdk::rte_flow_error = unsafe { mem::zeroed() };
412    unsafe {
413        let ret = dpdk::rte_flow_validate(
414            port.id.raw(),
415            attr.raw() as *const _,
416            pattern_rules.as_ptr(),
417            action.rules.as_ptr(),
418            &mut error as *mut _,
419        );
420        if ret != 0 {
421            error!("Default drop rule failed validation.");
422            let msg: &CStr = CStr::from_ptr(error.message);
423            bail!(HardwareFilterError::Validation {
424                lpattern: LayeredPattern::new(),
425                reason: msg.to_str().unwrap().to_string()
426            });
427        } else {
428            let ret = dpdk::rte_flow_create(
429                port.id.raw(),
430                attr.raw() as *const _,
431                pattern_rules.as_ptr(),
432                action.rules.as_ptr(),
433                &mut error as *mut _,
434            );
435            if ret.is_null() {
436                error!("Default drop rule failed creation.");
437                let msg: &CStr = CStr::from_ptr(error.message);
438                bail!(HardwareFilterError::Creation {
439                    lpattern: LayeredPattern::new(),
440                    reason: msg.to_str().unwrap().to_string()
441                });
442            } else {
443                info!("Created hardware flow rule to drop ethernet traffic by default.");
444            }
445        }
446    }
447    Ok(())
448}

450/// Flush all flow rules associated with port
451pub(crate) fn flush_rules(port: &Port) {
452    info!("Flushing flow rules on Port {}", port.id);
453    unsafe {
454        let mut error: dpdk::rte_flow_error = mem::zeroed();
455        let ret = dpdk::rte_flow_flush(port.id.raw(), &mut error as &mut _);
456        if ret != 0 {
457            let msg: &CStr = CStr::from_ptr(error.message);
458            panic!(
459                "Flush rules failed. Port in inconsistent state, please restart: {}",
460                msg.to_str().unwrap()
461            );
462        }
463    }
